{{- /* The Keep's live chart (PRODUCT.md, "Live plot"). At build time every metric is fetched once and embedded as a compact snapshot: bucket indices and values in two arrays, so a week of 15-minute buckets for both devices is a few kB gzipped. assets/js/keep-live.js draws from the snapshot at once, then replaces it with live data. If the endpoint is down, at build or in the browser, the chart keeps the last data it has and says how old it is. A metric that failed at build is simply absent from the snapshot; scripts/check.sh warns on that. */ -}} {{- $cfg := site.Params.keepLive -}} {{- $days := int $cfg.days -}} {{- $snap := dict -}} {{- range $m := slice "battery" "temp" "rssi" -}} {{- $series := dict -}} {{- $meta := dict -}} {{- range $d := $cfg.devices -}} {{- $u := printf "%s/public/v1/%s?device=%s&days=%d" $cfg.api $m $d.id $days -}} {{- with try (resources.GetRemote $u) -}} {{- with .Err -}} {{- warnf "keep-live: %s: %s" $u . -}} {{- else with .Value -}} {{- $j := . | transform.Unmarshal -}} {{- $start := (time.AsTime $j.start).Unix -}} {{- /* Two flat arrays: Hugo's append cannot build a slice of pairs. */ -}} {{- $is := slice -}} {{- $vs := slice -}} {{- range $j.points -}} {{- $is = $is | append (div (sub (time.AsTime .ts).Unix $start) $j.bucket_s) -}} {{- $vs = $vs | append (div (math.Round (mul .value 100)) 100) -}} {{- end -}} {{- $series = merge $series (dict $d.id (dict "i" $is "v" $vs)) -}} {{- $meta = dict "start" $j.start "end" $j.end "bucket" $j.bucket_s -}} {{- end -}} {{- end -}} {{- end -}} {{- with $meta -}} {{- $snap = merge $snap (dict $m (merge . (dict "series" $series))) -}} {{- end -}} {{- end -}} {{- $labels := slice -}} {{- range $cfg.devices }}{{ $labels = $labels | append (strings.TrimPrefix "Device " .label) }}{{ end -}}
